from flask import Flask, request, jsonify

приложение = Flask(__name__)

# ================= ЛОГИКА СОВЕТОВ =================

def дать_совет(тема, сообщение):

    сообщение = сообщение.lower()

    if тема == "вокал":
        return (
            "🎤 Совет по вокалу:\n"
            "- Срежь низ до 80 Hz\n"
            "- Компрессия: Ratio 3:1\n"
            "- Attack 10-20 ms\n"
            "- Release 50-100 ms\n"
            "- Добавь де-эссер на 5-8 kHz\n"
            "- Легкий реверб 10-15%"
        )

    elif тема == "барабаны":
        return (
            "🥁 Совет по барабанам:\n"
            "- Усиль атаку 4-6 kHz\n"
            "- Подрежь мутные 200-300 Hz\n"
            "- Параллельная компрессия\n"
            "- Добавь сатурацию для плотности"
        )

    elif тема == "мастеринг":
        return (
            "🎛 Совет по мастерингу:\n"
            "- Цель LUFS: -14 для стриминга\n"
            "- True Peak не выше -1 dB\n"
            "- Лимитер в конце цепочки\n"
            "- Контролируй низ до 40 Hz"
        )

    else:
        return "Опиши проблему подробнее."

# ================= API =================

@приложение.route("/спросить", methods=["POST"])
def спросить():
    данные = request.json
    тема = данные.get("тема")
    текст = данные.get("текст")
    ответ = дать_совет(тема, текст)
    return jsonify({"ответ": ответ})
